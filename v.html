// 1. 修改後的選章 Function function selectChap(chap) { selectedChap = chap; const trigger = document.getElementById('chapTrigger'); trigger.classList.remove('placeholder'); document.getElementById('chapTriggerText').textContent = `第 ${chap} 章`; trigger.setAttribute('aria-label', `章數：第 ${chap} 章，點擊更改`); document.querySelectorAll('#chapList .dropdown-item').forEach(el => { const v = parseInt(el.dataset.value); el.classList.toggle('selected', v === chap); el.setAttribute('aria-selected', v === chap ? 'true' : 'false'); }); closeAllDropdowns(); // 渲染經文內容 renderScripture(selectedBook, selectedChap); // 啟動全新的「語音無縫倒數」 initAutoPlay(); } // 2. 全新升級：利用語音隊列破解封鎖的倒數機制 function initAutoPlay() { // 先清空任何舊的計時器與正在播放的語音 if (autoPlayTimer) { clearInterval(autoPlayTimer); autoPlayTimer = null; } if ('speechSynthesis' in window) window.speechSynthesis.cancel(); ttsPlaying = true; ttsPaused = false; updateTtsUI(); const statusBar = document.getElementById('ttsStatusBar'); const statusLabel = document.getElementById('ttsLabel'); const currentVerse = document.getElementById('ttsCurrentVerse'); statusBar.style.display = 'block'; statusBar.className = 'paused'; // 橙黃色等待外觀 statusLabel.textContent = '🕒 自動朗讀預備'; currentVerse.textContent = '3 秒後將自動播放...'; // 滾動畫面到經文開始處 document.getElementById('scriptureSection').scrollIntoView({ behavior: 'smooth', block: 'start' }); if (!('speechSynthesis' in window) || !ttsVerses.length) return; // 建立倒數語音物件 (在 click 事件內同步建立並播放，100% 破解封鎖) const uReady = new SpeechSynthesisUtterance("準備朗讀"); const u3 = new SpeechSynthesisUtterance("三"); const u2 = new SpeechSynthesisUtterance("二"); const u1 = new SpeechSynthesisUtterance("一"); // 統一設定廣東話配音同語速 [uReady, u3, u2, u1].forEach(u => { u.lang = 'zh-HK'; u.rate = ttsRate; if (ttsVoice) u.voice = ttsVoice; }); // 聽覺與視覺同步：當語音讀到該字時，畫面的字體跟住變 uReady.onstart = () => { currentVerse.textContent = "準備朗讀..."; document.getElementById('a11yAnnouncer').textContent = "準備自動朗讀經文"; }; u3.onstart = () => { currentVerse.textContent = "🕒 3 秒..."; }; u2.onstart = () => { currentVerse.textContent = "🕒 2 秒..."; }; u1.onstart = () => { currentVerse.textContent = "🕒 1 秒..."; }; // 【核心無縫接軌】當數到「一」完結的瞬間，立刻開始讀第一節經文 u1.onend = () => { if (!ttsPlaying) return; ttsIndex = 0; updateTtsUI(); speakCurrent(); // 呼叫原本讀經文的內容 }; // 安全計：萬一倒數語音在某些舊機出錯，直接切換去讀經文 uReady.onerror = u3.onerror = u2.onerror = u1.onerror = (e) => { if (e.error === 'interrupted' || e.error === 'canceled') return; ttsIndex = 0; speakCurrent(); }; // 一口氣將倒數指令塞入瀏覽器（利用當前的 Click 授權一併解鎖） window.speechSynthesis.speak(uReady); window.speechSynthesis.speak(u3); window.speechSynthesis.speak(u2); window.speechSynthesis.speak(u1); } // 3. 優化控制按鈕：如果在倒數期間點擊了「立即播放」，直接跳過倒數開播 function ttsToggle() { if (!('speechSynthesis' in window)) { alert('你的裝置不支援語音朗讀。'); return; } if (!ttsVerses.length) return; // 如果在語音倒數預備期間點擊了按鈕，直接取消倒數，立刻讀經文 const statusLabel = document.getElementById('ttsLabel').textContent; if (statusLabel.includes('預備')) { window.speechSynthesis.cancel(); ttsStartDirectly(); return; } if (!ttsPlaying) { ttsIndex = 0; ttsPlaying = true; ttsPaused = false; updateTtsUI(); speakCurrent(); } else if (!ttsPaused) { window.speechSynthesis.pause(); ttsPaused = true; updateTtsUI(); } else { window.speechSynthesis.resume(); ttsPaused = false; updateTtsUI(); } }
